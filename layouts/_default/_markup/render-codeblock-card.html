{{- /* Render hook: turn a ```flashcard fenced code block into a reveal card. Usage in markdown: ```flashcard Q: Question text. A: Answer text (may span multiple lines / markdown). ``` The block must contain a `Q:` line and an `A:` line. Everything from `Q:` up to the `A:` line is the question; everything after is the answer. Both are rendered as markdown, so lists, bold, code, and math all work — just leave a blank line before block-level markdown (lists, etc.). */ -}} {{- $raw := trim .Inner "\n" -}} {{- $q := "" -}} {{- $a := "" -}} {{- if hasPrefix $raw "Q:" -}} {{- $raw = trim (strings.TrimPrefix "Q:" $raw) " " -}} {{- end -}} {{- $parts := split $raw "\nA:" -}} {{- if ge (len $parts) 2 -}} {{- $q = trim (index $parts 0) " \n" -}} {{- $a = trim (index $parts 1) " \n" -}} {{- else -}} {{- errorf "flashcard: block must contain a `Q:` line and an `A:` line (%s)" .Position -}} {{- end -}} {{- /* Emit the shared CSS only once per rendered page. */ -}} {{- if not (.Page.Scratch.Get "flashcardAssetsLoaded") -}} {{- .Page.Scratch.Set "flashcardAssetsLoaded" true -}} {{- end -}}
{{ $q | markdownify }}
{{ $a | markdownify }}
