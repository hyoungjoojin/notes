{{- /* flashcard shortcode — click a card to reveal its answer. Usage in markdown: {{< flashcard >}} [Q] Question text here. [A] Answer text here (may span multiple lines). {{< /flashcard >}} Everything after [Q] up to [A] is the question; everything after [A] is the answer. Both are rendered as markdown. */ -}} {{- $raw := trim .Inner "\n" -}} {{- $q := "" -}} {{- $a := "" -}} {{- if and (in $raw "[Q]") (in $raw "[A]") -}} {{- $afterQ := index (split $raw "[Q]") 1 -}} {{- $parts := split $afterQ "[A]" -}} {{- $q = trim (index $parts 0) " \n" -}} {{- $a = trim (index $parts 1) " \n" -}} {{- else -}} {{- errorf "flashcard shortcode: content must contain both [Q] and [A] markers (%s)" .Position -}} {{- end -}} {{- /* Emit the shared CSS/JS only once per rendered page. */ -}} {{- if not (.Page.Scratch.Get "flashcardAssetsLoaded") -}} {{- .Page.Scratch.Set "flashcardAssetsLoaded" true -}} {{- end -}}
{{ $q | markdownify }}
{{ $a | markdownify }}
